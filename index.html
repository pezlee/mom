請貼上 Google 試算表「發布到網路（CSV 格式）」網址，或 Google Apps Script 部署的 /exec 網址：

進階：若需在網頁修改數值並反向寫回 Google Sheet（點此展開 Apps Script 程式碼）

注意：若目前填入的是「發布到網路的 CSV（/pub?output=csv）」只能讀取；要將網頁編輯/新增的數值寫回 Google Sheet，必須貼上Apps Script 部署後的 /exec 網址，且部署時「誰可以存取」選所有人（每次改腳本都要建立「新版本」部署）。 

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  var data = sheet.getDataRange().getDisplayValues();
  var payload = JSON.stringify(data);
  if (e && e.parameter && e.parameter.callback) {
    return ContentService.createTextOutput(e.parameter.callback + '(' + payload + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(payload)
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '';
    if (!raw && e && e.parameter && e.parameter.payload) {
      raw = e.parameter.payload;
    }
    var parsed = JSON.parse(raw);
    var rows = Array.isArray(parsed) ? parsed : (parsed.rows || []);
    if (!rows.length) {
      return ContentService.createTextOutput(JSON.stringify({ ok: false, error: '無可寫入的資料列' }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
    var maxCols = rows.reduce(function(m, r) { return Math.max(m, r.length); }, 1);
    var normalized = rows.map(function(r) {
      var copy = r.slice(0, maxCols);
      while (copy.length < maxCols) copy.push('');
      return copy;
    });
    sheet.clearContents();
    sheet.getRange(1, 1, normalized.length, maxCols).setValues(normalized);
    return ContentService.createTextOutput(JSON.stringify({ ok: true, count: normalized.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
